[Numlock]

-/usr/local/bin/numlock.sh

  #!/bin/bash

  for tty in /dev/tty{1..6}
  do
      /usr/bin/setleds -D +num < "$tty";
  done

   ----> Make the script EXECUTABLE!

-/etc/systemd/system/numlock.service

   [Unit]
   Description=numlock

   [Service]
   ExecStart=/usr/local/bin/numlock.sh
   StandardInput=tty
   RemainAfterExit=yes

   [Install]
   WantedBy=multi-user.target

   ---> Then sudo systemctl enable numlock.service
